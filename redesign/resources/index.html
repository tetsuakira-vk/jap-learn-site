Setup — do this once

  1. Download Scriptable from the App Store — it's free
  2. Open Scriptable, tap + top right → paste the script you want → rename it → tap Done
  3. Long-press your home screen → tap + → search Scriptable → choose a size
  4. Long-press the widget → Edit Widget → set Script to the one you saved → set When Interacting to Run Script

Script 1 — Word of the Day

WordOfTheDay.js Small widget 
// Japanese Word of the Day — japaneseunlocked.com
const WORDS = [
  {jp:"水",rd:"みず",en:"water"},{jp:"火",rd:"ひ",en:"fire"},
  {jp:"山",rd:"やま",en:"mountain"},{jp:"川",rd:"かわ",en:"river"},
  {jp:"空",rd:"そら",en:"sky"},{jp:"海",rd:"うみ",en:"sea"},
  {jp:"風",rd:"かぜ",en:"wind"},{jp:"花",rd:"はな",en:"flower"},
  {jp:"食べる",rd:"たべる",en:"to eat"},{jp:"飲む",rd:"のむ",en:"to drink"},
  {jp:"見る",rd:"みる",en:"to see"},{jp:"話す",rd:"はなす",en:"to speak"},
  {jp:"行く",rd:"いく",en:"to go"},{jp:"来る",rd:"くる",en:"to come"},
  {jp:"大きい",rd:"おおきい",en:"big"},{jp:"小さい",rd:"ちいさい",en:"small"},
  {jp:"楽しい",rd:"たのしい",en:"fun"},{jp:"新しい",rd:"あたらしい",en:"new"},
  {jp:"学校",rd:"がっこう",en:"school"},{jp:"友達",rd:"ともだち",en:"friend"},
  {jp:"先生",rd:"せんせい",en:"teacher"},{jp:"電車",rd:"でんしゃ",en:"train"},
  {jp:"日本語",rd:"にほんご",en:"Japanese"},{jp:"勉強",rd:"べんきょう",en:"study"},
  {jp:"心",rd:"こころ",en:"heart"},{jp:"夢",rd:"ゆめ",en:"dream"},
];
const d = new Date();
const seed = d.getFullYear()*10000+(d.getMonth()+1)*100+d.getDate();
const word = WORDS[seed % WORDS.length];
const w = new ListWidget();
w.backgroundColor = new Color("#0e0e10");
w.setPadding(14,16,12,16);
const lbl = w.addText("WORD OF THE DAY");
lbl.font = Font.boldSystemFont(7); lbl.textColor = new Color("#c0392b"); lbl.leftAlignText();
w.addSpacer(8);
const jp = w.addText(word.jp);
jp.font = Font.boldSystemFont(42); jp.textColor = Color.white();
jp.leftAlignText(); jp.minimumScaleFactor = 0.4;
const rd = w.addText(word.rd);
rd.font = Font.systemFont(13); rd.textColor = new Color("#ffffff80"); rd.leftAlignText();
w.addSpacer(6);
const en = w.addText(word.en);
en.font = Font.mediumSystemFont(15); en.textColor = Color.white(); en.leftAlignText();
w.addSpacer();
const foot = w.addText("japaneseunlocked.com");
foot.font = Font.systemFont(7); foot.textColor = new Color("#ffffff30"); foot.leftAlignText();
Script.setWidget(w); Script.complete();
if(config.runsInApp) w.presentSmall();
